    private fun captureC11Music(name: String) {
        RuntimeEnvironment.setFontScale(1.3f)
        val screen = openSettings()
        ReflectionHelpers.setField(screen, "settingsCategory", SettingsCategory.VEHICLE)
        ReflectionHelpers.callInstanceMethod<Unit>(screen, "render")
        layoutRoot(screen)
        val title = texts(screen).first { it.text == screen.getString(R.string.settings_c11_music_guard) }
        val control = generateSequence(title.parent as ViewGroup) { it.parent as? ViewGroup }
            .first { group -> descendants(group).any { it is TextView && it.text == screen.getString(R.string.settings_c11_music_access) } }
        assertFalse(C11MusicGuard.enabled(context))
        val bitmap = Bitmap.createBitmap(control.width, minOf(control.height, 1800), Bitmap.Config.ARGB_8888)
        try {
            control.draw(Canvas(bitmap))
            val file = Paths.get("build/reports/c11-ui/$name.png")
            Files.createDirectories(file.parent)
            Files.newOutputStream(file).use { assertTrue(bitmap.compress(Bitmap.CompressFormat.PNG, 100, it)) }
        } finally { bitmap.recycle() }
    }

    private fun captureC11Profile(name: String, cluster: Boolean) {
        RuntimeEnvironment.setFontScale(1.3f)
        org.robolectric.shadows.ShadowBuild.setModel("msmnile_gvmq for arm64")
        val manager = context.getSystemService(android.hardware.display.DisplayManager::class.java)
        val displayId = org.robolectric.shadows.ShadowDisplayManager.addDisplay("w1920dp-h720dp-mdpi", 5)
        shadowOf(manager.getDisplay(displayId)).apply {
            setName("C11 instrument (simulated)")
            setFlags(android.view.Display.FLAG_PRESENTATION)
        }
        try {
            val screen = openSettings()
            if (cluster) {
                ReflectionHelpers.setField(screen, "settingsCategory", SettingsCategory.ADVANCED)
                ReflectionHelpers.setField(screen, "settingsSectionFilter", setOf(SettingsSection.CLUSTER_MAP))
            }
            else ReflectionHelpers.setField(screen, "page", "home")
            ReflectionHelpers.callInstanceMethod<Unit>(screen, "render")
            layoutRoot(screen)
            if (cluster) {
                val title = texts(screen).first { it.text.toString().startsWith(screen.getString(R.string.settings_c11_cluster_display)) }
                val scroll = ReflectionHelpers.getField<ScrollView>(screen, "rootScroll")
                val bounds = Rect(); title.getDrawingRect(bounds)
                scroll.offsetDescendantRectToMyCoords(title, bounds)
                scroll.scrollTo(0, (bounds.top - 16).coerceAtLeast(0))
                assertTrue(title.getGlobalVisibleRect(Rect()))
                assertNull(C11ClusterRoute.findDisplay(context))
                assertFalse(AdbClusterRouter.enabled(context))
                // Capture the C11 control itself; full-root software rendering can retain the pre-scroll frame.
                val control = title.parent as ViewGroup
                val controlBitmap = Bitmap.createBitmap(control.width, minOf(control.height, 1600), Bitmap.Config.ARGB_8888)
                try {
                    val canvas = Canvas(controlBitmap)
                    canvas.translate(0f, -(title.top - 24).coerceAtLeast(0).toFloat())
                    control.draw(canvas)
                    val file = Paths.get("build/reports/c11-ui/$name-control.png")
                    Files.createDirectories(file.parent)
                    Files.newOutputStream(file).use { assertTrue(controlBitmap.compress(Bitmap.CompressFormat.PNG, 100, it)) }
                } finally { controlBitmap.recycle() }
            }
            val root = screen.findViewById<ViewGroup>(android.R.id.content).getChildAt(0)
            val bitmap = Bitmap.createBitmap(root.width, root.height, Bitmap.Config.ARGB_8888)
            try {
                root.draw(Canvas(bitmap))
                val output = Paths.get("build/reports/c11-ui/$name.png")
                Files.createDirectories(output.parent)
                Files.newOutputStream(output).use { assertTrue(bitmap.compress(Bitmap.CompressFormat.PNG, 100, it)) }
            } finally { bitmap.recycle() }
        } finally { org.robolectric.shadows.ShadowDisplayManager.removeDisplay(displayId) }
    }
